    protected ImportStatistics $statistics;

    protected ExistingVolunteersCache $existing;

    public function __construct()
    {
        $this->statistics = new ImportStatistics();

        $lookup = new LookupCache();

        $this->mapper = new VolunteerExcelMapper(
            $lookup,
            $this->statistics
        );

        $this->existing = new ExistingVolunteersCache();
    }

    /**
     * معالجة صف من الإكسيل
     */
    public function model(array $row)
    {
        // dd($row);
        // dd(array_keys($row));
        try {

            $data = $this->mapper->map(
                $row,
                $this->statistics->totalRows() + 1
            );
            // dd($data);
            if (!$data) {
                return null;
            }

            /*
            |--------------------------------------------------------------------------
            | Insert / Update Counters
            |--------------------------------------------------------------------------
            */

            if (
                $this->existing->exists(
                    $data['military_number']
                )
            ) {

                $this->statistics->updated();

            } else {

                $this->statistics->inserted();

                $this->existing->add(
                    $data['military_number']
                );
            }

            return new Volunteer($data);
            // Volunteer::create($data);

            // dd('saved');

        } catch (Throwable $e) {

            $this->statistics->database(
                $this->statistics->totalRows(),
                $e->getMessage()
            );

            return null;
        }
    }

    /**
     * مفتاح الـ Upsert

    protected function qualified(array $row): bool
    {
        return isset($row['mohl']) &&
            trim($row['mohl']) === 'مؤهل';
    }

    protected function notQualified(array $row): bool
    {
        return isset($row['ghyr_mohl']) &&
            trim($row['ghyr_mohl']) === 'غير مؤهل';
    }

    protected function checkLookup(
        int $row,
        string $field,
        ?string $value,
        ?int $id
    ): void {

        if (!blank($value) && !$id) {

            $this->stats->lookup(
                $row,
                $field,
                trim($value)
            );

        }
    }

    protected function string($value): ?string
    {
        if (blank($value)) {
            return null;
        }

        return trim($value);
    }

    protected function excelDate($value): ?string
    {
        if (blank($value)) {
            return null;
        }

        try {

            if (is_numeric($value)) {

                return Date::excelToDateTimeObject($value)
                    ->format('Y-m-d');
            }

            return Carbon::parse($value)
                ->format('Y-m-d');

        } catch (\Throwable) {

            return null;
        }
    }

    {
        return $this->stats;

class ExistingVolunteersCache
{
    /**
     * military_number => id
     */
    protected array $volunteers = [];

    public function __construct()
    {
        $this->load();
    }

    protected function load(): void
    {
        $this->volunteers = Volunteer::query()
            ->select('id', 'military_number')
            ->pluck('id', 'military_number')
            ->toArray();
    }

    /**
     * هل المتطوع موجود؟
     */
    public function exists(?string $militaryNumber): bool
    {
        if (blank($militaryNumber)) {
            return false;
        }

        return isset(
            $this->volunteers[
                trim($militaryNumber)
            ]
        );
    }

    /**
     * الحصول على ID
     */
    public function id(?string $militaryNumber): ?int
    {
        if (!$this->exists($militaryNumber)) {
            return null;
        }

        return $this->volunteers[
            trim($militaryNumber)
        ];
    }

    /**
     * إضافة متطوع أثناء الاستيراد
     */
    public function add(
        string $militaryNumber,
        ?int $id = null
    ): void {

        $this->volunteers[
            trim($militaryNumber)
        ] = $id;
    }

    /**
     * عدد المتطوعين الموجودين
     */
    public function count(): int
    {
        return count($this->volunteers);
    }

    /**
     * إعادة تحميل الكاش
     */
    public function refresh(): void
    {
        $this->load();
    }
}
